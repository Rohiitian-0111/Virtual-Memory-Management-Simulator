cmake_minimum_required(VERSION 3.16)
project(VirtualMemoryManagementSimulator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(vm_core src/vm.cpp)
target_include_directories(vm_core PUBLIC include)
add_executable(vm_simulator src/main.cpp)
target_link_libraries(vm_simulator PRIVATE vm_core)
enable_testing()
add_executable(vm_tests tests.cpp)
target_link_libraries(vm_tests PRIVATE vm_core)
add_test(NAME vm_tests COMMAND vm_tests)
